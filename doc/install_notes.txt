# the following is what it took to get TASSI working on a CentOS 8 system as of May 2025

sudo dnf install -y ansible
sudo dnf install -y epel-release
sudo dnf install -y wireguard-tools
sudo dnf install -y python38 python38-devel
sudo alternatives --install /usr/bin/unversioned-python python /usr/bin/python3.8 1
sudo alternatives --set python /usr/bin/python3.8

# tofu
wget --secure-protocol=TLSv1_2 --https-only https://get.opentofu.org/install-opentofu.sh -O install-opentofu.sh
chmod +x install-opentofu.sh
sudo ./install-opentofu.sh --install-method rpm
rm -f install-opentofu.sh

sudo dnf install -y python3-libvirt
sudo dnf install -y python3-lxml

sudo dnf install -y libvirt libvirt-daemon-kvm
sudo systemctl enable --now libvirtd
sudo dnf install -y virt-install

# ensure NFS can handle locking which qemu needs
sudo systemctl status rpc-statd

# ensure libvirt participates in ns lookup
sudo dnf install -y libvirt-nss
sudo sed -i '/^hosts:/ s/\(.*\)\bdns\b/\1libvirt libvirt_guest dns/' /etc/nsswitch.conf
